cmake_minimum_required(VERSION 3.25)
project(Mriya VERSION 1.0.0 LANGUAGES C ASM)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

option(MR_SYSTEM_SDL "Link against an SDL3 already installed on the system" OFF)

# ---------------------------------------------------------------------------
# SDL3 (zlib license). Statically linked so the screensaver is a single file.
# ---------------------------------------------------------------------------
if(MR_SYSTEM_SDL)
  find_package(SDL3 REQUIRED CONFIG)
else()
  include(FetchContent)
  set(SDL_SHARED OFF CACHE BOOL "" FORCE)
  set(SDL_STATIC ON  CACHE BOOL "" FORCE)
  set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  set(SDL_UNINSTALL OFF CACHE BOOL "" FORCE)
  # Only what a screensaver needs: video + OpenGL + events + timers, and the
  # joystick for flying it.
  foreach(sub AUDIO GPU RENDER CAMERA HAPTIC SENSOR DIALOG POWER OPENGLES TRAY)
    set(SDL_${sub} OFF CACHE BOOL "" FORCE)
  endforeach()
  set(SDL_OPENGL ON CACHE BOOL "" FORCE)
  set(SDL_JOYSTICK ON CACHE BOOL "" FORCE)
  set(SDL_HIDAPI ON CACHE BOOL "" FORCE)
  FetchContent_Declare(SDL3
    URL      https://github.com/libsdl-org/SDL/releases/download/release-3.4.16/SDL3-3.4.16.tar.gz
    URL_HASH SHA256=7322236cd12090c3eb40b9728be4d49c76f66ad17d04369584d4ecad5cf77c68
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
  FetchContent_MakeAvailable(SDL3)
  # See The Black Hole: no dynamic API, per-function sections, so the linker
  # drops everything a screensaver never calls.
  if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_definitions(SDL3-static PRIVATE SDL_THREAD_SAFETY_ANALYSIS)
    target_compile_options(SDL3-static PRIVATE -ffunction-sections -fdata-sections)
  endif()
endif()

# ---------------------------------------------------------------------------
# The aircraft. tools/meshpack.c welds, smooths, labels and packs the GLB at
# build time, and the result is linked straight into the executable with
# .incbin - a 5 MB C array would take the compiler longer than everything else
# put together.
# ---------------------------------------------------------------------------
add_executable(meshpack tools/meshpack.c)
if(NOT WIN32)
  target_link_libraries(meshpack PRIVATE m)
endif()
set(MR_GEN ${CMAKE_CURRENT_BINARY_DIR}/gen)
set(MR_GLB ${CMAKE_CURRENT_SOURCE_DIR}/res/models/Mriya.glb)
set(MR_MESH ${MR_GEN}/mriya.mesh)
set(MR_DECALS ${CMAKE_CURRENT_SOURCE_DIR}/res/textures/decals.tex)
add_custom_command(
  OUTPUT  ${MR_MESH}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${MR_GEN}
  COMMAND $<TARGET_FILE:meshpack> ${MR_GLB} ${MR_MESH}
  DEPENDS meshpack ${MR_GLB}
  COMMENT "Packing the aircraft mesh")
set(MR_BLOBS_S ${MR_GEN}/blobs.S)
file(WRITE ${MR_BLOBS_S}.in
"/* Generated by CMakeLists.txt - do not edit. */
#define BLOB(name, path) \\
    .section .rodata ; .balign 16 ; .global name ; name: ; .incbin path ; \\
    .global name##_end ; name##_end: ; .byte 0
BLOB(mr_mesh_blob, \"@MR_MESH@\")
BLOB(mr_decal_blob, \"@MR_DECALS@\")
#ifdef __ELF__
.section .note.GNU-stack,\"\",@progbits
#endif
")
configure_file(${MR_BLOBS_S}.in ${MR_BLOBS_S} @ONLY)
set_source_files_properties(${MR_BLOBS_S} PROPERTIES OBJECT_DEPENDS "${MR_MESH};${MR_DECALS}")

# ---------------------------------------------------------------------------
# Shaders are embedded as C byte arrays at build time (see cmake/embed.cmake).
# common.glsl is prepended to every one of them at run time.
# ---------------------------------------------------------------------------
set(MR_SHADERS
  src/shaders/common.glsl src/shaders/atmosphere.glsl src/shaders/terrain_fn.glsl
  src/shaders/cloud_fn.glsl
  src/shaders/post.vert src/shaders/blit.frag src/shaders/blur.frag src/shaders/bright.frag
  src/shaders/lut_transmittance.frag src/shaders/lut_multiscatter.frag
  src/shaders/lut_skyview.frag src/shaders/lut_aerial.frag
  src/shaders/noise3d.frag src/shaders/weather.frag src/shaders/cloudshadow.frag
  src/shaders/sky.frag src/shaders/terrain.vert src/shaders/terrain.frag
  src/shaders/aircraft.vert src/shaders/aircraft.frag src/shaders/shadow.frag
  src/shaders/clouds.frag src/shaders/cloudtaa.frag src/shaders/composite.frag src/shaders/probe.frag
  src/shaders/precip.vert src/shaders/precip.frag
  src/shaders/bolt.vert src/shaders/bolt.frag
  src/shaders/lights.vert src/shaders/lights.frag
  src/shaders/trail.vert src/shaders/trail.frag
  src/shaders/lum.frag src/shaders/adapt.frag src/shaders/final.frag src/shaders/present.frag src/shaders/msresolve.frag
  src/shaders/hud.vert src/shaders/hud.frag)
set(MR_SHADER_ABS)
foreach(s ${MR_SHADERS})
  list(APPEND MR_SHADER_ABS ${CMAKE_CURRENT_SOURCE_DIR}/${s})
endforeach()
set(MR_SHADERS_GEN ${MR_GEN}/shaders_gen.c)
add_custom_command(
  OUTPUT  ${MR_SHADERS_GEN}
  COMMAND ${CMAKE_COMMAND} -DOUT=${MR_SHADERS_GEN} "-DINPUTS=${MR_SHADER_ABS}"
          -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/embed.cmake
  DEPENDS ${MR_SHADER_ABS} cmake/embed.cmake
  COMMENT "Embedding GLSL shaders")

# ---------------------------------------------------------------------------
# The screensaver executable
# ---------------------------------------------------------------------------
set(MR_SOURCES
  src/main.c src/app.c src/settings.c src/render.c src/gl_loader.c
  src/model.c src/flight.c src/weather.c src/camera.c src/terrain.c src/hud.c src/trails.c src/realtime.c
  ${MR_SHADERS_GEN} ${MR_BLOBS_S})
if(WIN32)
  list(APPEND MR_SOURCES src/platform_win32.c res/win32/mriya.rc)
else()
  list(APPEND MR_SOURCES src/platform_linux.c)
endif()

add_executable(Mriya ${MR_SOURCES})
target_include_directories(Mriya PRIVATE src third_party res/win32)
target_link_libraries(Mriya PRIVATE SDL3::SDL3)
target_compile_definitions(Mriya PRIVATE
  MR_VERSION="${PROJECT_VERSION}" $<$<CONFIG:Debug>:MR_DEBUG=1>)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(Mriya PRIVATE $<$<COMPILE_LANGUAGE:C>:-Wall -Wextra -Wshadow
    -Wno-unused-parameter -ffunction-sections -fdata-sections>)
  target_link_options(Mriya PRIVATE $<$<CONFIG:Release>:-Wl,--gc-sections>)
endif()

if(WIN32)
  # A Windows screensaver is a GUI-subsystem exe with the .scr extension.
  set_target_properties(Mriya PROPERTIES WIN32_EXECUTABLE TRUE SUFFIX ".scr")
  target_link_libraries(Mriya PRIVATE comctl32 comdlg32 opengl32 gdi32 user32 advapi32 shell32 winhttp)
  if(CMAKE_C_COMPILER_ID MATCHES "GNU")
    # Static CRT, stripped, hardening flags set explicitly (see The Black Hole
    # for why each one is there).
    target_link_options(Mriya PRIVATE -static -static-libgcc
      $<$<CONFIG:Release>:-s>
      -Wl,--dynamicbase -Wl,--nxcompat -Wl,--high-entropy-va
      -Wl,--enable-reloc-section
      -Wl,--major-subsystem-version,6 -Wl,--minor-subsystem-version,1
      -Wl,--major-os-version,6 -Wl,--minor-os-version,1)
    target_link_options(meshpack PRIVATE -static -static-libgcc)
  endif()
  target_compile_definitions(Mriya PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN)
else()
  target_link_libraries(Mriya PRIVATE m)
  set_target_properties(Mriya PROPERTIES OUTPUT_NAME mriya)
  include(GNUInstallDirs)
  set(MR_XSS_HACKDIR "${CMAKE_INSTALL_LIBEXECDIR}/xscreensaver" CACHE PATH "XScreenSaver hack directory")
  set(MR_XSS_CONFDIR "${CMAKE_INSTALL_DATADIR}/xscreensaver/config" CACHE PATH "XScreenSaver config directory")
  install(TARGETS Mriya RUNTIME DESTINATION ${MR_XSS_HACKDIR})
  install(FILES res/linux/mriya.xml DESTINATION ${MR_XSS_CONFDIR})
endif()
